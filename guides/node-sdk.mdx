## Meanwhile — use fetch

```typescript
const PROMPTWALL_API_KEY = process.env.PROMPTWALL_API_KEY!;
const BASE = "https://api.prompt-wall.com";

async function verify(prompt: string, answer: string, toolResult?: any) {
  const res = await fetch(`${BASE}/v1/verify`, {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${PROMPTWALL_API_KEY}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({ prompt, answer, tool_result: toolResult }),
  });
  if (!res.ok) throw new Error(`PromptWall ${res.status}`);
  return res.json();
}
```
